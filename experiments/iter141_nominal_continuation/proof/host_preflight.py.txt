import json
import os
from pathlib import Path
import shutil
import subprocess
import time
report=dict(host_utc=time.time(),hostname=os.uname().nodename,free_gib=shutil.disk_usage('/datasets/nuscenes-full').free/2**30,affinity=sorted(os.sched_getaffinity(0)))
assert report['hostname'].split('.')[0]=='sentinel-gpu'
for name,cmd in {'hugsim':['git','-C','/opt/sentinel-stack/HUGSIM','rev-parse','HEAD'],'uniad':['git','-C','/opt/sentinel-stack/UniAD_SIM','rev-parse','HEAD'],'containers':['docker','ps','--format','{{.ID}} {{.Names}}'],'gpu_processes':['nvidia-smi','--query-compute-apps=pid,process_name','--format=csv,noheader'],'processes':['ps','-eo','pid,etimes,pcpu,comm']}.items():
 r=subprocess.run(cmd,capture_output=True,text=True,timeout=30);report[name]=dict(returncode=r.returncode,stdout=r.stdout,stderr=r.stderr)
assert report['hugsim']['stdout'].strip()=='62c690d39fd90020e68a196bd8bcc1c4d4191f2e'
assert report['uniad']['stdout'].strip()=='5fb279e39912a5ac7f58e00d56b065cadcd0a749'
assert report['containers']['returncode']==0 and not report['containers']['stdout'].strip()
assert report['gpu_processes']['returncode']==0 and not report['gpu_processes']['stdout'].strip()
assert report['free_gib']>=100
remaining=int(1791568619.076531-time.time())
assert remaining>1230
r=subprocess.run(['systemd-run','--unit=sentinel-iter141-budget','--on-active='+str(remaining)+'s','/sbin/shutdown','-h','now'],capture_output=True,text=True)
report['shutdown_timer']=dict(returncode=r.returncode,seconds=remaining,stdout=r.stdout,stderr=r.stderr)
print(json.dumps(report,indent=2));assert r.returncode==0
