import json,os,subprocess
from pathlib import Path
root=Path('/datasets/nuscenes-full/hugsim/iter142_control_20261009');output=Path('/datasets/nuscenes-full/hugsim/outputs/iter142-20261009')
assert list(output.iterdir())==[]
argv=['/usr/bin/env','PYTHONPATH='+str(root/'packet'),'python3',str(root/'run_episode.py'),'--help']
r=subprocess.run(argv,capture_output=True,text=True,timeout=45)
commands={}
for name,cmd in [('containers',['docker','ps','-q']),('gpu',['nvidia-smi','--query-compute-apps=pid','--format=csv,noheader'])]:
 x=subprocess.run(cmd,capture_output=True,text=True,timeout=30);commands[name]=dict(returncode=x.returncode,stdout=x.stdout,stderr=x.stderr)
print(json.dumps(dict(argv=argv,returncode=r.returncode,stdout=r.stdout,stderr=r.stderr,output_entries=[p.name for p in output.iterdir()],commands=commands),indent=2))
assert r.returncode==0 and 'reference_b' in r.stdout and 'nominal_a' in r.stdout and 'nominal_b' in r.stdout
assert list(output.iterdir())==[] and all(x['returncode']==0 and not x['stdout'].strip() for x in commands.values())
