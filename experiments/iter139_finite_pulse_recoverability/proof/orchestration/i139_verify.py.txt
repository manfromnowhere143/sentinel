import json,sys
sys.path.insert(0,'/tmp')
from i139_ops import ssh,PROOF
arm=sys.argv[1];assert arm in ['reference_a','reference_b','pulse_0_1','pulse_0_2','pulse_2_1','pulse_2_2','pulse_4_1','pulse_4_2']
script='''import json,os,subprocess
from pathlib import Path
root=Path('/datasets/nuscenes-full/hugsim/iter139_control_20261009');output=Path('/datasets/nuscenes-full/hugsim/outputs/iter139-20261009');arm=ARM
cmd=['/opt/sentinel-stack/HUGSIM/.pixi/envs/default/bin/python','-m','experiments.iter139_finite_pulse_recoverability.verify_live',str(output/'reference_a/scene-0411_hard_00')]
if arm!='reference_a':cmd.append(str(output/arm/'scene-0411_hard_00'))
env={**os.environ,'PYTHONPATH':str(root/'packet'),'OPENBLAS_NUM_THREADS':'1','OMP_NUM_THREADS':'1','MKL_NUM_THREADS':'1'}
r=subprocess.run(cmd,capture_output=True,text=True,env=env,timeout=300)
print(json.dumps(dict(returncode=r.returncode,stdout=r.stdout,stderr=r.stderr),indent=2))
if r.returncode==0:
 report=json.loads(r.stdout);assert report['status']=='validated'
 with (root/'validations'/(arm+'.json')).open('x') as f:json.dump(report,f,indent=2,sort_keys=True)
assert r.returncode==0
'''.replace('ARM',repr(arm))
r=ssh(arm+'_verify',script,timeout=330);assert r.returncode==0
wrapper=json.loads(r.stdout);assert wrapper['returncode']==0
(PROOF/'analysis').mkdir(exist_ok=True)
(PROOF/'analysis'/(arm+'.json')).write_text(wrapper['stdout'])
