import datetime,json,subprocess,sys,time
from pathlib import Path
R=Path('/Users/danielwahnich/workspace/sentinel');P=R/'experiments/iter143_model_contract/proof'
label=sys.argv[1];args=sys.argv[2:];start=time.time()
r=subprocess.run(args,cwd=R,capture_output=True)
for suffix,raw in [('stdout',r.stdout),('stderr',r.stderr)]:
 with (P/(label+'.'+suffix)).open('xb') as f:f.write(raw)
with (P/(label+'.json')).open('x') as f:json.dump(dict(argv=args,cwd=str(R),start_utc=datetime.datetime.fromtimestamp(start,datetime.timezone.utc).isoformat(),elapsed_seconds=time.time()-start,returncode=r.returncode),f,indent=2);f.write('\n')
print(label,r.returncode,r.stdout.decode()[-2500:],r.stderr.decode()[-1000:]);sys.exit(r.returncode)
